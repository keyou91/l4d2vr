# Compile the production functions against scripted input without pulling in
# the game's rendering and engine dependencies.
function(extract_mirrored_analog_fixture source_text begin_marker end_marker result_name)
    string(FIND "${source_text}" "${begin_marker}" begin_index)
    if(begin_index LESS 0)
        message(FATAL_ERROR "Mirrored analog fixture start marker missing: ${begin_marker}")
    endif()
    string(SUBSTRING "${source_text}" ${begin_index} -1 source_tail)
    string(FIND "${source_tail}" "${end_marker}" end_index)
    if(end_index LESS 0)
        message(FATAL_ERROR "Mirrored analog fixture end marker missing: ${end_marker}")
    endif()
    string(SUBSTRING "${source_tail}" 0 ${end_index} fixture_text)
    set(${result_name} "${fixture_text}" PARENT_SCOPE)
endfunction()

function(extract_throw_lifecycle_function source_text function_name result_name)
    string(REGEX MATCH "[^\r\n;{}]+[ \t]${function_name}\\([^;{}]*\\)[ \t\r\n]*\\{" function_header "${source_text}")
    if(NOT function_header)
        message(FATAL_ERROR "Throw lifecycle function definition missing: ${function_name}")
    endif()
    string(FIND "${source_text}" "${function_header}" function_start)
    string(SUBSTRING "${source_text}" ${function_start} -1 function_tail)
    string(FIND "${function_tail}" "{" brace_start)
    math(EXPR cursor "${brace_start} + 1")
    string(LENGTH "${function_tail}" tail_length)
    set(depth 1)
    set(quote "")
    set(line_comment FALSE)
    set(block_comment FALSE)
    while(depth GREATER 0 AND cursor LESS tail_length)
        string(SUBSTRING "${function_tail}" ${cursor} 1 character)
        math(EXPR pair_remaining "${tail_length} - ${cursor}")
        if(pair_remaining GREATER 1)
            string(SUBSTRING "${function_tail}" ${cursor} 2 pair)
        else()
            set(pair "")
        endif()
        if(line_comment)
            if(character STREQUAL "\n")
                set(line_comment FALSE)
            endif()
        elseif(block_comment)
            if(pair STREQUAL "*/")
                set(block_comment FALSE)
                math(EXPR cursor "${cursor} + 1")
            endif()
        elseif(NOT quote STREQUAL "")
            if(character STREQUAL "\\")
                math(EXPR cursor "${cursor} + 1")
            elseif(character STREQUAL quote)
                set(quote "")
            endif()
        elseif(pair STREQUAL "//")
            set(line_comment TRUE)
            math(EXPR cursor "${cursor} + 1")
        elseif(pair STREQUAL "/*")
            set(block_comment TRUE)
            math(EXPR cursor "${cursor} + 1")
        elseif(character STREQUAL "\"" OR character STREQUAL "'")
            set(quote "${character}")
        elseif(character STREQUAL "{")
            math(EXPR depth "${depth} + 1")
        elseif(character STREQUAL "}")
            math(EXPR depth "${depth} - 1")
        endif()
        math(EXPR cursor "${cursor} + 1")
    endwhile()
    if(NOT depth EQUAL 0)
        message(FATAL_ERROR "Throw lifecycle function has unbalanced braces: ${function_name}")
    endif()
    string(SUBSTRING "${function_tail}" 0 ${cursor} function_text)
    set(${result_name} "${function_text}" PARENT_SCOPE)
endfunction()
